   1  import { expm1, log1p } from '../../numeric';
   2  import type { Factor } from '../types';
   3  import type { FusedErrorRates, FusionRule, PreparedFusion } from './fusion-rule';
   4  
   5  /**
   6   * Правило «И» (СРО1, формула (2)): злоумышленник должен пройти все включённые факторы.
   7   * FAR = ∏ FAR_i, FRR = 1 − ∏(1 − FRR_i). Вычисление в лог-пространстве (ADR-0008).
   8   */
   9  export class AndFusionRule implements FusionRule {
  10    readonly id = 'AND' as const;
  11  
  12    prepare(factors: readonly Factor[]): PreparedFusion {
  13      const logFar = factors.map((f) => Math.log(f.far));
  14      const logSurv = factors.map((f) => log1p(-f.frr));
  15      return {
  16        combine(indices: readonly number[]): FusedErrorRates {
  17          let lf = 0;
  18          let ls = 0;
  19          for (const i of indices) {
  20            lf += logFar[i]!;
  21            ls += logSurv[i]!;
  22          }
  23          // ls = 0 (все FRR_i = 0) → FRR = 0, а не −0.
  24          return { far: Math.exp(lf), logFar: lf, frr: ls === 0 ? 0 : -expm1(ls) };
  25        },
  26      };
  27    }
  28  }

   8  export const MAX_SWEEP_POINTS = 500;
   9  
  10  export interface BudgetSweepRange {
  11    readonly from: number;
  12    readonly to: number;
  13    readonly step: number;
  14  }
  15  
  16  export interface BudgetSweepPoint {
  17    readonly budget: number;
  18    readonly codes: readonly string[] | null;
  19    readonly objective: number | null;
  20  }
  21  
  22  export interface BudgetBreakpoint extends BudgetSweepPoint {
  23    /** E_new / E_old относительно предыдущего оптимума — «цена» изменения бюджета. */
  24    readonly ratioToPrevious: number | null;
  25  }
  26  
  27  export interface BudgetSweepResult {
  28    readonly points: readonly BudgetSweepPoint[];
  29    readonly breakpoints: readonly BudgetBreakpoint[];
  30  }
  31  
  32  /**
  33   * Анализ чувствительности к бюджету (док. 02, §2.2.4): наборы, допустимые без учёта бюджета, сортируются
  34   * по стоимости; при движении B вверх поддерживается лучший набор среди C(x) ≤ B. O(2ⁿ·n + F·log F + точки).
  35   */
  36  export function budgetSweep(problem: FactorSelectionProblem, range: BudgetSweepRange): BudgetSweepResult {
  37    const relaxed: FactorSelectionProblem = {
  38      ...problem,
  39      constraints: { ...problem.constraints, budget: null },
  40    };
  41    validateProblem(relaxed);
  42    const n = problem.factors.length;
  43    if (n > EXHAUSTIVE_MAX_FACTORS) {
  44      throw new ProblemTooLargeError(`анализ бюджета поддерживает не более ${EXHAUSTIVE_MAX_FACTORS} факторов`);
  45    }
  46    assertDomain(
  47      isFiniteNumber(range.from) && range.from >= 0,
  48      'начало диапазона должно быть ≥ 0',
  49      'sweep.from',
  50    );
  51    assertDomain(
  52      isFiniteNumber(range.to) && range.to >= range.from,
  53      'конец диапазона должен быть ≥ начала',
  54      'sweep.to',
  55    );
  56    assertDomain(isFiniteNumber(range.step) && range.step > 0, 'шаг должен быть > 0', 'sweep.step');
  57    const count = Math.floor((range.to - range.from) / range.step + 1e-9) + 1;
  58    assertDomain(count <= MAX_SWEEP_POINTS, `не более ${MAX_SWEEP_POINTS} точек`, 'sweep.step');
  59  
  60    const evaluator = new SubsetEvaluator(relaxed);
  61    const codeOf = (i: number) => problem.factors[i]!.code;
  62    const candidates: SubsetMetrics[] = [];
  63    for (let mask = 1; mask < 2 ** n; mask++) {
  64      const m = evaluator.evaluate(indicesFromMask(mask, n));
  65      if (evaluator.isFeasible(m)) candidates.push(m);
  66    }
  67    candidates.sort((a, b) => a.cost - b.cost);
  68  
  69    const points: BudgetSweepPoint[] = [];
  70    let pointer = 0;
  71    let best: SubsetMetrics | null = null;
  72    for (let k = 0; k < count; k++) {
  73      const budget = Number((range.from + k * range.step).toFixed(12));
  74      while (pointer < candidates.length && leq(candidates[pointer]!.cost, budget)) {
  75        const c = candidates[pointer++]!;
  76        if (best === null || compareSubsets(c, best, codeOf) < 0) best = c;
  77      }
  78      points.push({
  79        budget,
  80        codes: best === null ? null : best.indices.map(codeOf),
  81        objective: best === null ? null : best.objective,
  82      });
  83    }
  84  
  85    const breakpoints: BudgetBreakpoint[] = [];
  86    let prev: BudgetSweepPoint | null = null;
  87    for (const p of points) {
  88      const key = p.codes?.join('+') ?? null;
  89      const prevKey = prev ? (prev.codes?.join('+') ?? null) : undefined;
  90      if (prev === null || key !== prevKey) {
  91        const prevObjective = prev?.objective ?? null;
  92        const ratio = prevObjective !== null && p.objective !== null ? p.objective / prevObjective : null;
  93        breakpoints.push({ ...p, ratioToPrevious: ratio });
  94      }
  95      prev = p;
  96    }
  97    return { points, breakpoints };
  98  }

   1  import { EPS, expm1, log1p } from '../../numeric';
   2  import { compareSubsets, SubsetEvaluator } from '../subset-evaluator';
   3  import { TopK } from '../top-k';
   4  import type { FactorSelectionProblem, SelectionResult, SolveOptions, SubsetMetrics } from '../types';
   5  import { MAX_FACTORS, validateProblem } from '../validation';
   6  import { type FactorSelectionSolver, resolveRankingLimit } from './factor-selection-solver';
   7  import type { MilpConstraint, MilpModel, MilpPort, MilpSolution } from './milp-port';
   8  
   9  /** Запас на численный допуск MILP при вычислении нижней оценки (консервативно уменьшает Λ). */
  10  const LAMBDA_SAFETY = 1e-9;
  11  /** Допуск «ничьей» по Λ в лексикографическом шаге и запас границы Парето по L (в сторону включения наборов). */
  12  const TIE_TOLERANCE = 1e-9;
  13  const DEFAULT_MAX_ITERATIONS = 10_000;
  14  
  15  export interface CertifiedIlpOptions {
  16    readonly maxIterations?: number;
  17  }
  18  
  19  const slackRhs = (rhs: number) => rhs + EPS * Math.max(1, Math.abs(rhs));
  20  
  21  /**
  22   * Решатель B (док. 02, §2.2.3) — метод СРО1: логарифмирование сводит задачу (2)–(3) к ЦЛП.
  23   *
  24   * Обе ошибки набора — монотонные функции двух ЛИНЕЙНЫХ форм от x:
  25   *   L(x) = Σ ln(FAR_i)·x_i,             FAR(x) = e^{L(x)}
  26   *   Λ(x) = −Σ ln(1 − FRR_i)·x_i ≥ 0,    FRR(x) = 1 − e^{−Λ(x)}   (точнее приближения Σ FRR_i·x_i из статьи)
  27   * поэтому E(x) = cFA·e^{L} + cFR·(1 − e^{−Λ}) возрастает по обеим и оптимум лежит на фронте Парето (Λ, L).
  28   *
  29   * Фронт обходится методом ε-ограничений:
  30   *   1) x̂ = argmin Λ  при  L(x) ≤ L_bound  (+ ограничения (3) и отсечения уже просмотренных наборов);
  31   *   2) среди наборов с Λ ≤ Λ(x̂) берётся argmin L — эффективная точка x̂′ (разрешение «ничьих» по Λ);
  32   *   3) L_bound ← L(x̂′): все наборы с бóльшим L доминируются x̂′ и исключаются.
  33   * Каждый следующий шаг даёт Λ не меньше предыдущего, поэтому LB = cFR·(1 − e^{−Λ(x̂)}) — нижняя граница E
  34   * для всех ещё не просмотренных наборов. Остановка: LB ≥ лучшего E (оптимум доказан) или наборов не осталось.
  35   */
  36  export class CertifiedIlpSolver implements FactorSelectionSolver {
  37    readonly id = 'ilp' as const;
  38    readonly maxFactors = MAX_FACTORS;
  39  
  40    constructor(
  41      private readonly milp: MilpPort,
  42      private readonly options: CertifiedIlpOptions = {},
  43    ) {}
  44  
  45    async solve(problem: FactorSelectionProblem, options?: SolveOptions): Promise<SelectionResult> {
  46      validateProblem(problem);
  47      const n = problem.factors.length;
  48      const limit = resolveRankingLimit(options);
  49      const maxIterations = Math.min(this.options.maxIterations ?? DEFAULT_MAX_ITERATIONS, 2 ** n);
  50      const evaluator = new SubsetEvaluator(problem);
  51      const codeOf = (i: number) => problem.factors[i]!.code;
  52      const top = new TopK<SubsetMetrics>(limit, (a, b) => compareSubsets(a, b, codeOf));
  53  
  54      const base = this.baseModel(problem);
  55      const lambda = problem.factors.map((f) => -log1p(-f.frr));
  56      const logFar = problem.factors.map((f) => Math.log(f.far));
  57      const lambdaOf = (indices: readonly number[]) => indices.reduce((s, i) => s + lambda[i]!, 0);
  58      const toIndices = (s: Extract<MilpSolution, { status: 'optimal' }>) =>
  59        s.values.flatMap((v, i) => (v > 0.5 ? [i] : []));
  60  
  61      const cuts: MilpConstraint[] = [];
  62      const seen = new Set<string>();
  63      let paretoBound: number | null = null;
  64      let best: SubsetMetrics | null = null;
  65      let iterations = 0;
  66      let certifiedBound: number | null = null;
  67      let proven = false;
  68  
  69      while (iterations < maxIterations) {
  70        const constraints = [...base.constraints, ...cuts];
  71        if (paretoBound !== null) {
  72          constraints.push({ name: 'pareto', coeffs: logFar, sense: '<=', rhs: paretoBound + TIE_TOLERANCE });
  73        }
  74        // Шаг 1: минимальное Λ среди оставшихся наборов.
  75        const byLambda = await this.milp.solve({ ...base, objective: lambda, constraints });
  76        if (byLambda.status === 'infeasible') {
  77          proven = true; // недоминируемых наборов больше нет
  78          break;
  79        }
  80        iterations++;
  81        const first = evaluator.evaluate(toIndices(byLambda));
  82        const lambdaK = lambdaOf(first.indices);
  83        const lowerBound = problem.costs.cFR * -expm1(-Math.max(0, lambdaK - LAMBDA_SAFETY));
  84        if (best !== null && lowerBound >= best.objective) {
  85          certifiedBound = lowerBound;
  86          proven = true;
  87          break;
  88        }
  89        // Шаг 2: среди «ничьих» по Λ — набор с минимальным FAR (эффективная точка фронта).
  90        const byFar = await this.milp.solve({
  91          ...base,
  92          objective: logFar,
  93          constraints: [
  94            ...constraints,
  95            { name: 'lex', coeffs: lambda, sense: '<=', rhs: lambdaK + TIE_TOLERANCE * Math.max(1, lambdaK) },
  96          ],
  97        });
  98        const efficient = byFar.status === 'optimal' ? evaluator.evaluate(toIndices(byFar)) : first;
  99  
 100        for (const m of [first, efficient]) {
 101          const key = m.indices.join(',');
 102          if (seen.has(key)) continue;
 103          seen.add(key);
 104          // Точная проверка допустимости — арбитр (допуски MILP могут пропустить граничный набор).
 105          if (evaluator.isFeasible(m)) {
 106            top.offer(m);
 107            if (best === null || compareSubsets(m, best, codeOf) < 0) best = m;
 108          }
 109          cuts.push(this.noGoodCut(m.indices, n, cuts.length));
 110        }
 111        // Шаг 3: сдвиг границы Парето — только по точно допустимому набору (иначе можно отсечь допустимые).
 112        if (evaluator.isFeasible(efficient)) paretoBound = efficient.logFar;
 113      }
 114  
 115      const ranking = top.toArray().map((m) => evaluator.toSolution(m));
 116      return {
 117        status: best === null ? 'infeasible' : 'optimal',
 118        optimum: best === null ? null : evaluator.toSolution(best),
 119        ranking,
 120        rankingComplete: false,
 121        stats: { factorsCount: n, subsetsTotal: 2 ** n - 1, feasibleCount: null, rejectedBy: null },
 122        solver: {
 123          used: 'ilp',
 124          certificate: {
 125            iterations,
 126            lowerBound: certifiedBound,
 127            bestObjective: best?.objective ?? null,
 128            proven,
 129          },
 130        },
 131      };
 132    }
 133  
 134    private baseModel(problem: FactorSelectionProblem): MilpModel {
 135      const { factors, constraints: c } = problem;
 136      const n = factors.length;
 137      const constraints: MilpConstraint[] = [
 138        { name: 'nonempty', coeffs: factors.map(() => 1), sense: '>=', rhs: 1 },
 139      ];
 140      if (c.latencyMaxMs !== null) {
 141        constraints.push({
 142          name: 'latency',
 143          coeffs: factors.map((f) => f.latencyMs),
 144          sense: '<=',
 145          rhs: slackRhs(c.latencyMaxMs),
 146        });
 147      }
 148      if (c.budget !== null) {
 149        constraints.push({
 150          name: 'budget',
 151          coeffs: factors.map((f) => f.cost),
 152          sense: '<=',
 153          rhs: slackRhs(c.budget),
 154        });
 155      }
 156      if (c.minPadFactors > 0) {
 157        constraints.push({
 158          name: 'pad',
 159          coeffs: factors.map((f) => (f.hasPad ? 1 : 0)),
 160          sense: '>=',
 161          rhs: c.minPadFactors,
 162        });
 163      }
 164      if (c.farMax !== null) {
 165        // Точная линейная форма ограничения FAR(x) ≤ FAR_max в логарифмах.
 166        constraints.push({
 167          name: 'far',
 168          coeffs: factors.map((f) => Math.log(f.far)),
 169          sense: '<=',
 170          rhs: Math.log(c.farMax) + EPS,
 171        });
 172      }
 173      return { numVars: n, objective: factors.map(() => 0), constraints };
 174    }
 175  
 176    /** Σ_{i∈x̂} (1 − x_i) + Σ_{i∉x̂} x_i ≥ 1  ⇔  Σ_{i∉x̂} x_i − Σ_{i∈x̂} x_i ≥ 1 − |x̂|. */
 177    private noGoodCut(indices: readonly number[], n: number, k: number): MilpConstraint {
 178      const selected = new Set(indices);
 179      const coeffs = Array.from({ length: n }, (_, i) => (selected.has(i) ? -1 : 1));
 180      return { name: `cut${k}`, coeffs, sense: '>=', rhs: 1 - indices.length };
 181    }
 182  }

  17   * Решатель A (док. 02, §2.2.2): полный перебор 2ⁿ − 1 непустых подмножеств с точной нелинейной E(x).
  18   * Эталон точности. Для каждой маски метрики считаются заново в фиксированном порядке индексов —
  19   * без накопления погрешности при инкрементальных обновлениях.
  20   */
  21  export class ExhaustiveSolver implements FactorSelectionSolver {
  22    readonly id = 'exhaustive' as const;
  23    readonly maxFactors = EXHAUSTIVE_MAX_FACTORS;
  24  
  25    solve(problem: FactorSelectionProblem, options?: SolveOptions): Promise<SelectionResult> {
  26      try {
  27        return Promise.resolve(this.solveSync(problem, options));
  28      } catch (err) {
  29        return Promise.reject(err instanceof Error ? err : new Error(String(err)));
  30      }
  31    }
  32  
  33    solveSync(problem: FactorSelectionProblem, options?: SolveOptions): SelectionResult {
  34      validateProblem(problem);
  35      const n = problem.factors.length;
  36      if (n > this.maxFactors) {
  37        throw new ProblemTooLargeError(
  38          `полный перебор поддерживает не более ${this.maxFactors} факторов (передано ${n})`,
  39        );
  40      }
  41      const limit = resolveRankingLimit(options);
  42      const evaluator = new SubsetEvaluator(problem);
  43      const total = 2 ** n - 1;
  44      const rejectedBy: Record<ConstraintKind, number> = { latency: 0, budget: 0, pad: 0, far: 0 };
  45      const codeOf = (i: number) => problem.factors[i]!.code;
  46      const top = new TopK<SubsetMetrics>(limit, (a, b) => compareSubsets(a, b, codeOf));
  47      let feasibleCount = 0;
  48  
  49      for (let mask = 1; mask <= total; mask++) {
  50        const m = evaluator.evaluate(indicesFromMask(mask, n));
  51        const v = evaluator.violations(m);
  52        let ok = true;
  53        for (const k of CONSTRAINT_KINDS) {
  54          if (v[k]) {
  55            rejectedBy[k]++;
  56            ok = false;
  57          }
  58        }
  59        if (ok) {
  60          feasibleCount++;
  61          top.offer(m);
  62        }
  63      }
  64  
  65      const ranking = top.toArray().map((m) => evaluator.toSolution(m));
  66  
  67      return {
  68        status: ranking.length > 0 ? 'optimal' : 'infeasible',
  69        optimum: ranking[0] ?? null,
  70        ranking,
  71        rankingComplete: feasibleCount <= limit,
  72        stats: { factorsCount: n, subsetsTotal: total, feasibleCount, rejectedBy },
  73        solver: { used: 'exhaustive', certificate: null },
  74      };
  75    }
  76  }

  13    readonly latency: boolean;
  14    readonly budget: boolean;
  15    readonly pad: boolean;
  16    readonly far: boolean;
  17  }
  18  
  19  /**
  20   * Единственная ответственность — вычислить метрики подмножества и проверить допустимость (ограничения (3) СРО1).
  21   * Используется обоими решателями, поэтому их результаты сравнимы побитово.
  22   */
  23  export class SubsetEvaluator {
  24    private readonly fusion: PreparedFusion;
  25    private readonly logFarMax: number | null;
  26  
  27    constructor(
  28      readonly problem: FactorSelectionProblem,
  29      fusionRule: FusionRule = new AndFusionRule(),
  30    ) {
  31      this.fusion = fusionRule.prepare(problem.factors);
  32      this.logFarMax = problem.constraints.farMax === null ? null : Math.log(problem.constraints.farMax);
  33    }
  34  
  35    evaluate(indices: readonly number[]): SubsetMetrics {
  36      const { factors, costs } = this.problem;
  37      const fused = this.fusion.combine(indices);
  38      let latencyMs = 0;
  39      let cost = 0;
  40      let padCount = 0;
  41      let frrSum = 0;
  42      for (const i of indices) {
  43        const f = factors[i]!;
  44        latencyMs += f.latencyMs;
  45        cost += f.cost;
  46        frrSum += f.frr;
  47        if (f.hasPad) padCount++;
  48      }
  49      return {
  50        indices,
  51        far: fused.far,
  52        logFar: fused.logFar,
  53        frr: fused.frr,
  54        frrSum,
  55        latencyMs,
  56        cost,
  57        padCount,
  58        objective: costs.cFA * fused.far + costs.cFR * fused.frr,
  59      };
  60    }
  61  
  62    violations(m: SubsetMetrics): Violations {
  63      const c = this.problem.constraints;
  64      return {
  65        latency: c.latencyMaxMs !== null && !leq(m.latencyMs, c.latencyMaxMs),
  66        budget: c.budget !== null && !leq(m.cost, c.budget),
  67        pad: m.padCount < c.minPadFactors,
  68        far: this.logFarMax !== null && !leqLog(m.logFar, this.logFarMax),
  69      };
  70    }
  71  
  72    isFeasible(m: SubsetMetrics): boolean {
  73      if (m.indices.length === 0) return false;
  74      const v = this.violations(m);
  75      return !(v.latency || v.budget || v.pad || v.far);
  76    }
  77  
  78    toSolution(m: SubsetMetrics): SubsetSolution {
  79      const c = this.problem.constraints;
  80      const slack = {
  81        latencyMs: c.latencyMaxMs === null ? null : c.latencyMaxMs - m.latencyMs,
  82        budget: c.budget === null ? null : c.budget - m.cost,
  83        logFar: this.logFarMax === null ? null : this.logFarMax - m.logFar,
  84        minPadFactors: c.minPadFactors > 0 ? m.padCount - c.minPadFactors : null,
  85      };
  86      const active: ConstraintName[] = [];
  87      if (slack.latencyMs !== null && isTight(slack.latencyMs, c.latencyMaxMs!)) active.push('latencyMaxMs');
  88      if (slack.budget !== null && isTight(slack.budget, c.budget!)) active.push('budget');
  89      if (slack.logFar !== null && isTight(slack.logFar, 1)) active.push('farMax');
  90      if (slack.minPadFactors === 0) active.push('minPadFactors');
  91      return {
  92        codes: m.indices.map((i) => this.problem.factors[i]!.code),
  93        far: m.far,
  94        frr: m.frr,
  95        latencyMs: m.latencyMs,
  96        cost: m.cost,
  97        padCount: m.padCount,
  98        objective: m.objective,
  99        activeConstraints: active,
 100        slack,
 101      };
 102    }
 103  }

 105  export const CONSTRAINT_KINDS: readonly ConstraintKind[] = ['latency', 'budget', 'pad', 'far'];

 107  /**
 108   * Детерминированный порядок наборов (NFR-5): E ↑, FAR ↑, стоимость ↑, задержка ↑, затем коды лексикографически.
 109   */
 110  export function compareSubsets(a: SubsetMetrics, b: SubsetMetrics, codeOf: (i: number) => string): number {
 111    if (a.objective !== b.objective) return a.objective - b.objective;
 112    if (a.far !== b.far) return a.far - b.far;
 113    if (a.cost !== b.cost) return a.cost - b.cost;
 114    if (a.latencyMs !== b.latencyMs) return a.latencyMs - b.latencyMs;
 115    const ka = a.indices.map(codeOf).join('+');
 116    const kb = b.indices.map(codeOf).join('+');
 117    return ka < kb ? -1 : ka > kb ? 1 : 0;
 118  }

   1  /** Ограниченный упорядоченный список лучших K элементов (вставка бинарным поиском). */
   2  export class TopK<T> {
   3    private readonly items: T[] = [];
   4  
   5    constructor(
   6      private readonly k: number,
   7      private readonly compare: (a: T, b: T) => number,
   8    ) {}
   9  
  10    offer(item: T): void {
  11      const items = this.items;
  12      if (items.length === this.k && this.compare(item, items[items.length - 1] as T) >= 0) return;
  13      let lo = 0;
  14      let hi = items.length;
  15      while (lo < hi) {
  16        const mid = (lo + hi) >>> 1;
  17        if (this.compare(items[mid] as T, item) <= 0) lo = mid + 1;
  18        else hi = mid;
  19      }
  20      items.splice(lo, 0, item);
  21      if (items.length > this.k) items.pop();
  22    }
  23  
  24    toArray(): T[] {
  25      return [...this.items];
  26    }
  27  }

   1  /**
   2   * Бисекция корня непрерывной функции на [a, b] при f(a)·f(b) ≤ 0.
   3   * Останавливается, когда интервал перестаёт сжиматься в арифметике double.
   4   */
   5  export function bisect(f: (x: number) => number, a: number, b: number, maxIter = 200): number {
   6    let lo = a;
   7    let hi = b;
   8    let flo = f(lo);
   9    if (flo === 0) return lo;
  10    const fhi = f(hi);
  11    if (fhi === 0) return hi;
  12    if (Math.sign(flo) === Math.sign(fhi)) {
  13      throw new RangeError('bisect: на концах интервала функция одного знака');
  14    }
  15    for (let i = 0; i < maxIter; i++) {
  16      const mid = lo + (hi - lo) / 2;
  17      if (mid <= lo || mid >= hi) break;
  18      const fm = f(mid);
  19      if (fm === 0) return mid;
  20      if (Math.sign(fm) === Math.sign(flo)) {
  21        lo = mid;
  22        flo = fm;
  23      } else {
  24        hi = mid;
  25      }
  26    }
  27    return lo + (hi - lo) / 2;
  28  }

 145    const out = new Float64Array(a.length + b.length);
 146    let i = 0;
 147    let j = 0;
 148    let n = 0;
 149    let last = NaN;

 151      const takeA = j >= b.length || (i < a.length && a[i]! <= b[j]!);
 152      const v = takeA ? a[i++]! : b[j++]!;
 153      if (n === 0 || v !== last) {
 154        out[n++] = v;
 155        last = v;
 156      }

  16  export function stationarityQuadratic(input: GaussianThresholdInput): StationarityQuadratic {
  17    const { genuine: g, impostor: i, costs } = input;
  18    const a = 1 / (2 * g.std ** 2) - 1 / (2 * i.std ** 2);
  19    const b = i.mean / i.std ** 2 - g.mean / g.std ** 2;
  20    const c =
  21      g.mean ** 2 / (2 * g.std ** 2) -
  22      i.mean ** 2 / (2 * i.std ** 2) +
  23      Math.log(costs.cFA / costs.cFR) +
  24      Math.log(g.std / i.std);
  25    return { a, b, c };
  26  }
  27  
  28  /** Вещественные корни, численно устойчивая формула; при a ≈ 0 — линейный случай. */
  29  export function solveQuadratic({ a, b, c }: StationarityQuadratic, scale: number): number[] {
  30    if (Math.abs(a) <= 1e-12 * scale) {
  31      return b === 0 ? [] : [-c / b];
  32    }
  33    const d = b * b - 4 * a * c;
  34    if (d < 0) return [];
  35    const q = -0.5 * (b + (b >= 0 ? 1 : -1) * Math.sqrt(d));
  36    const roots = [q / a];
  37    if (q !== 0) roots.push(c / q);
  38    return roots.sort((x, y) => x - y);
  39  }
  40  
  41  /** τ_EER = (μ_I·σ_G + μ_G·σ_I) / (σ_G + σ_I) — замкнутая форма для нормальных распределений. */
  42  export function gaussianEerThreshold(input: Pick<GaussianThresholdInput, 'genuine' | 'impostor'>): number {
  43    const { genuine: g, impostor: i } = input;
  44    return (i.mean * g.std + g.mean * i.std) / (g.std + i.std);
  45  }
  46  
  47  /** Относительный выигрыш над вырожденным решением, ниже которого конечный порог не имеет практического смысла. */
  48  export const DEGENERATE_RELATIVE = 1e-9;
  49  
  50  interface Candidate {
  51    readonly tau: number | null;
  52    readonly far: number;
  53    readonly frr: number;
  54    readonly cost: number;
  55    readonly degenerate: OptimalPoint['degenerate'];
  56    readonly constrainedByFarMax: boolean;
  57  }
  58  
  59  export class GaussianAnalyticOptimizer implements ThresholdOptimizer<GaussianThresholdInput> {
  60    constructor(private readonly verifier: NumericVerifier = new NumericVerifier()) {}
  61  
  62    optimize(input: GaussianThresholdInput): ThresholdResult {
  63      validateNormal(input.genuine, 'genuine');
  64      validateNormal(input.impostor, 'impostor');
  65      validateCosts(input.costs);
  66      validateFarMax(input.farMax);
  67      assertDomain(
  68        input.genuine.mean > input.impostor.mean,
  69        'среднее оценок законных пользователей должно быть больше среднего злоумышленников',
  70        'genuine.mean',
  71      );
  72  
  73      const { costs } = input;
  74      const farMax = input.farMax ?? null;
  75      const model = new GaussianScoreModel(input.genuine, input.impostor);
  76      const point = (tau: number): OperatingPoint => {
  77        const far = model.far(tau);
  78        const frr = model.frr(tau);
  79        return { tau, far, frr, cost: costs.cFA * far + costs.cFR * frr };
  80      };
  81  
  82      const eer = point(gaussianEerThreshold(input));
  83  
  84      // Ограничение FAR(τ) ≤ FAR_max ⇔ τ ≥ τ_min (FAR монотонно убывает).
  85      const tauMin =
  86        farMax === null ? -Infinity : input.impostor.mean + input.impostor.std * SQRT2 * erfcinv(2 * farMax);
  87  
  88      const quad = stationarityQuadratic(input);
  89      const scale = 1 / (2 * input.genuine.std ** 2) + 1 / (2 * input.impostor.std ** 2);
  90      // E′(τ) имеет знак −Q(τ); локальный минимум E там, где Q убывает: Q′(τ) = 2aτ + b < 0.
  91      const localMinima = solveQuadratic(quad, scale).filter((t) => 2 * quad.a * t + quad.b < 0 && t >= tauMin);
  92  
  93      const candidates: Candidate[] = localMinima.map((t) => ({
  94        ...point(t),
  95        degenerate: null,
  96        constrainedByFarMax: false,
  97      }));
  98      if (farMax !== null) {
  99        candidates.push({ ...point(tauMin), degenerate: null, constrainedByFarMax: true });
 100      } else {
 101        candidates.push({
 102          tau: null,
 103          far: 1,
 104          frr: 0,
 105          cost: costs.cFA,
 106          degenerate: 'acceptAll',
 107          constrainedByFarMax: false,
 108        });
 109      }
 110      candidates.push({
 111        tau: null,
 112        far: 0,
 113        frr: 1,
 114        cost: costs.cFR,
 115        degenerate: 'rejectAll',
 116        constrainedByFarMax: false,
 117      });
 118  
 119      let best = candidates[0]!;
 120      for (const cand of candidates) if (cand.cost < best.cost) best = cand;
 121      // Корень, который лучше вырожденного решения меньше чем на DEGENERATE_RELATIVE, практически неотличим от него:
 122      // такой τ лежит в дальнем хвосте, где E(τ) плоская до машинной точности. Сообщаем вырожденное решение.
 123      const limit = candidates.filter((c) => c.degenerate !== null).reduce((a, b) => (b.cost < a.cost ? b : a));
 124      if (
 125        best.degenerate === null &&
 126        !best.constrainedByFarMax &&
 127        best.cost >= limit.cost * (1 - DEGENERATE_RELATIVE)
 128      ) {
 129        best = limit;
 130      }
 131  
 132      const optimal: OptimalPoint = {
 133        tau: best.tau,
 134        far: best.far,
 135        frr: best.frr,
 136        cost: best.cost,
 137        interval: null,
 138        degenerate: best.degenerate,
 139        constrainedByFarMax: best.constrainedByFarMax,
 140      };
 141  
 142      const verification =
 143        best.degenerate === null && best.tau !== null
 144          ? this.verifier.verify(model, costs, best.tau, tauMin)
 145          : { status: 'skipped' as const, numericTau: null, absDiff: null };
 146  
 147      const warnings: string[] = [];
 148      if (best.degenerate === 'acceptAll') {
 149        warnings.push(
 150          'При заданных стоимостях выгоднее допускать всех: порог не имеет смысла (cFA слишком мала).',
 151        );
 152      }
 153      if (best.degenerate === 'rejectAll') {
 154        warnings.push(
 155          'При заданных стоимостях выгоднее отвергать всех: порог не имеет смысла (cFR слишком мала).',
 156        );
 157      }
 158      if (farMax !== null && eer.far > farMax) {
 159        warnings.push('Точка равных ошибок не удовлетворяет ограничению FAR ≤ FAR_max.');
 160      }
 161      if (verification.status === 'mismatch') {
 162        warnings.push('Численная проверка не подтвердила аналитический оптимум.');
 163      }
 164  
 165      return {
 166        method: 'gaussian-analytic',
 167        eer,
 168        optimal,
 169        lossReduction: eer.cost > 0 ? 1 - optimal.cost / eer.cost : null,
 170        verification,
 171        warnings,
 172        sampleSizes: null,
 173      };
 174    }
 175  }

   1  import { bisect } from '../../numeric';
   2  import type { GaussianScoreModel } from '../gaussian-score-model';
   3  import type { Costs, ThresholdVerification } from '../types';
   4  
   5  export const VERIFICATION_TOLERANCE = 1e-7;
   6  const GRID_POINTS = 10_000;
   7  
   8  /**
   9   * Независимая проверка аналитического оптимума (ADR-0006):
  10   * 1) сетка из 10⁴ точек находит глобальный минимум E(τ) грубо;
  11   * 2) бисекция по производной E′(τ) = cFR·f_G(τ) − cFA·f_I(τ), вычисленной напрямую через плотности
  12   *    (без квадратного уравнения), уточняет его до машинной точности.
  13   */
  14  export class NumericVerifier {
  15    verify(
  16      model: GaussianScoreModel,
  17      costs: Costs,
  18      analyticTau: number,
  19      tauMin: number,
  20    ): ThresholdVerification {
  21      const { genuine: g, impostor: i } = model;
  22      const spread = Math.max(g.std, i.std);
  23      let lo = i.mean - 10 * i.std;
  24      let hi = Math.max(g.mean + 10 * g.std, analyticTau + spread);
  25      if (Number.isFinite(tauMin)) {
  26        lo = Math.max(lo, tauMin);
  27        hi = Math.max(hi, tauMin + 10 * spread);
  28      }
  29      const cost = (t: number) => costs.cFA * model.far(t) + costs.cFR * model.frr(t);
  30      const step = (hi - lo) / (GRID_POINTS - 1);
  31      let best = 0;
  32      let bestCost = Infinity;
  33      for (let k = 0; k < GRID_POINTS; k++) {
  34        const c = cost(lo + k * step);
  35        if (c < bestCost) {
  36          bestCost = c;
  37          best = k;
  38        }
  39      }
  40  
  41      let numericTau: number;
  42      if (best === 0 && Number.isFinite(tauMin)) {
  43        numericTau = lo; // минимум на границе τ_min
  44      } else {
  45        const derivative = (t: number) => costs.cFR * model.genuinePdf(t) - costs.cFA * model.impostorPdf(t);
  46        const a = lo + Math.max(0, best - 1) * step;
  47        const b = lo + Math.min(GRID_POINTS - 1, best + 1) * step;
  48        const da = derivative(a);
  49        const db = derivative(b);
  50        if (best > 0 && best < GRID_POINTS - 1 && da <= 0 && db >= 0 && (da !== 0 || db !== 0)) {
  51          numericTau = bisect(derivative, a, b);
  52        } else {
  53          numericTau = lo + best * step;
  54        }
  55      }
  56      const absDiff = Math.abs(numericTau - analyticTau);
  57      return { status: absDiff <= VERIFICATION_TOLERANCE ? 'ok' : 'mismatch', numericTau, absDiff };
  58    }
  59  }
